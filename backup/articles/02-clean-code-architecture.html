# fibonacci_generator.py
def fibonacci_stream(limit: int):
    """
    تولیدکننده دنباله فیبوناچی با پیچیدگی زمانی O(N)
    """
    a, b = 0, 1
    for _ in range(limit):
        yield a
        a, b = b, a + b

if __name__ == "__main__":
    result = list(fibonacci_stream(10))
    print(f"Fibonacci sequence: {result}")

قطعه کد تایپ‌اسکریپت و ری‌اکت هوک (REACT HOOK & CLIPBOARD)

نمونه‌ای از یک هوک مدرن در سمت فرانت‌اند برای مدیریت کپی در حافظه موقت سیستم: 

import { useState, useCallback } from 'react';

export function useClipboard(timeout = 2000) {
  const [hasCopied, setHasCopied] = useState(false);

  const copy = useCallback(async (text: string) => {
    try {
      await navigator.clipboard.writeText(text);
      setHasCopied(true);
      setTimeout(() => setHasCopied(false), timeout);
    } catch (err) {
      console.error('Clipboard copy failed:', err);
    }
  }, [timeout]);

  return { hasCopied, copy };
}
